# LightEval framework adapter requirements

# LightEval framework. 0.13.0 pins nltk==3.9.1 (42 advisories), so nltk is raised to
# 3.10.4 by requirements-overrides.txt. lighteval main is not pinned by git commit: it
# fixes none of what this adapter patches (LiteLLM loglikelihood, PIQA, TruthfulQA) and
# the requirements check (verify-requirements.yml) rejects VCS requirements.
lighteval==0.13.0
# LightEval 0.13.0 uses list_models(model_name=...), removed in Hub v2
huggingface-hub>=0.30.2,<2
pyarrow

# LiteLLM for endpoint support.
# 1.84.0 is affected by PYSEC-2026-4066; 1.102.2 is the newest release on the Red Hat
# package index (1.103.x is not there, verify-requirements.yml rejects it) that audits
# clean and does not cap filelock<4 (>=1.104 does, which keeps filelock on its
# vulnerable 3.x line). See DEPENDENCY-SECURITY.md. lighteval_logprob_patch.py turns
# litellm's response cache off (a cache hit on 1.103.9 needed orjson and soundfile).
litellm[caching]==1.102.2

# Required dependencies
requests==2.33.0
pydantic==2.13.5
# LightEval 0.13.0 passes strings to xxhash.xxh64(); xxhash 4.x requires bytes
xxhash>=3.0.0,<4

# Pin xxhash below v4 to avoid "Strings must be encoded before hashing"
# TypeError in lighteval's info_loggers.py (upstream bug)
xxhash<4

# EvalHub SDK (framework adapter with adapter extra only)
eval-hub-sdk[adapter]~=1.0.6

# CA bundle floor, matching the inspect adapter. requests/httpx trust certifi's
# cacert.pem rather than the system store; the downstream lock pins it exactly.
certifi>=2026.10.23
